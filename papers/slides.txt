\section*{Introduction}

\begin{frame}{Latent Factor Models: New Data, New Challenges}

\begin{itemize}

    \item Many latent factor models in the social sciences (e.g., ideal point models based on voting records, topic models for texts). 

    \item Unstructured data is plentiful and offers new research opportunities (e.g., texts, but also images, audio, and video recordings).

    \item But unstructured data often means:
    \begin{itemize}
        \item Large number of observations (\textbf{\textit{large n}}) 
        \item High dimensionality (\textbf{\textit{large p}}) 
        \item Observed metadata (\textbf{i.e., covariates, outcomes})
        \item Multiple modalities (\textbf{e.g., texts \textit{and} videos \textit{and} surveys}) 
        % pre-trained embeddings OR from scratch
    \end{itemize}

% Writing down a complex DGP is very easy (e.g., structural + multimodel), but estimating the model is hard. That's why we use neural networks.

% There are structural or multimodal latent factor models out there, but we propose a unified fast + structural + multimodal latent factor model.
% It allows both structured and unstructured data inputs.

% Why not Bertopic? Bertopic does not assume a DGP and has no theoretical guarantees, so typically, depending on the embeddings you use, you may not obtain interpretable topics.

% There's growing interest in the literature of understanding whether we can bridge advances in machine learning / deep learning and traditional, statistical models used in the social sciences. We are contributing to this strand of research by focusing on unsupervised learning problems.

% 

\end{itemize}

\end{frame}

\section*{Model}

\begin{frame}{The Model} \label{structural_model}

For observation $i \in \{1,...,N\}$:

\begin{equation*} \label{eq:deeplatent_normalized}
\begin{aligned}
    &\textbf{Latent factors:} 
    && z_i \sim p_{\psi}(z_i \mid x_i^p), \\[0.5em]
    &\textbf{Response variables:} 
    && w_{im} \sim p_{\theta_m}(w_{im} \mid z_i, x_i^c), \quad m = 1,\ldots,M, \\[0.5em]
    &\textbf{Auxiliary outcomes:} 
    && y_{ik} \sim p_{\theta_k}(y_{ik} \mid z_i, x_i^s), \quad k = 1,\ldots,K,
\end{aligned}
\end{equation*}
\begin{itemize}
    \item[] where:
    \begin{itemize}
        \item $z_i$ is a vector of $L$ latent factors.
        \item $w_{im}$ is a vector of response variables for each modality $m$.
        \item $y_{ik}$ is outcome $k$.
        \item $p_{\psi}(z_i \mid x_i^p)$ is a prior distribution.
        \item $x_i^p$, $x_i^c$, and $x_i^s$ are covariates that influence the latent, response, or outcome variables.
    \end{itemize}
\end{itemize}

% Mention that the main assumption is conditional independence between modalities/outcomes. Once we know the latent factor z, all these variables are assumed to be independent. 

\end{frame}

\begin{frame}{Special Case 1: Probabilistic PCA}

For observation $i \in \{1,...,N\}$:

\begin{equation*}
\begin{aligned}
    &\textbf{Latent factors:} 
    && z_i \sim \mathcal{N}(0, I), \\[0.75em]
    &\textbf{Response variables:} 
    && w_i \sim \mathcal{N}(W z_i + \mu,\ \sigma^2 I_D),\\[0.5em]
    &\textbf{Auxiliary outcomes:} 
    && --,
\end{aligned}
\end{equation*}

\begin{itemize}
    \item[] where:
    \begin{itemize}
        \item $z_i$: latent factors.
        \item $w_i$: observed vector.
        \item $W$: loading matrix.
        \item $\mu$: mean vector.
        \item $\sigma^2$: isotropic noise variance.
    \end{itemize}
\end{itemize}

% You can mention ideal-point models based on politicians' voting records as a classical use case.

\end{frame}

\begin{frame}{Special Case 2: A "Structural" Topic Model}

For observation $i \in \{1,...,N\}$:

\begin{equation*}
\begin{aligned}
    &\textbf{Latent factors:} 
    && z_i \sim \mathrm{LN}(\Gamma'X_i,\ \Sigma), \\[0.75em]
    &\textbf{Response variables:} 
    && w_i \sim \mathrm{Multinomial}(N_i,\ f(X_i, z_i)),\\[0.5em]
    &\textbf{Auxiliary outcomes:} 
    && --,
\end{aligned}
\end{equation*}

\begin{itemize}
    \item[] where:
    \begin{itemize}
        \item $X_i$ is a vector of covariates.
        \item $z_i$: topic proportions drawn from a logistic--normal.
        \item $w_i$: word-count vector.
        \item $N_i$: document length.
        \item $f(\cdot)$ is a function of doc-topic shares and covariates.
    \end{itemize}
\end{itemize}

% Mention here the STM by Stewart and co-authors, and explain briefly how covariates affect z_i (topic choice) (Eq 1), and how, conditional on topic choice z_i, covariates may also influence the words used to discuss a topic (topic slant) (Eq 2)

\end{frame}

\begin{frame}{Special Case 3: A "Supervised" Topic Model}

For observation $i \in \{1,...,N\}$:

\begin{equation*}
\begin{aligned}
    &\textbf{Latent factors:} 
    && z_i \sim \mathrm{LN}(0,\ I), \\[0.75em]
    &\textbf{Response variables:} 
    && w_i \sim \mathrm{Multinomial}(N_i,\ \Phi z_i),\\[0.5em]
    &\textbf{Auxiliary outcomes:} 
    && y_{i} = \beta' z_i + \varepsilon_i,
\end{aligned}
\end{equation*}

\begin{itemize}
    \item[] where:
    \begin{itemize}
        \item $z_i$: topic proportions drawn from a logistic--normal.
        \item $w_i$: word-count vector.
        \item $N_i$: document length.
        \item $\Phi$: topic–word matrix.
        \item $\beta$ captures the correlation between topics $z_i$ and $y_i$.
    \end{itemize}
\end{itemize}

% Many such supervised topic models are in the literature. They are particularly useful for creating interpretable predictions in the macroeconomics and conflict literatures (often based on newspaper articles). 

\end{frame}

% Nothing prevents you from doing much more complicated models of texts and images with transformer neural nets for the encoders/decoders (see next slide)

\begin{frame}{Special Case 4: An Image--Text Model}

For observation $i \in \{1,...,N\}$:

\begin{equation*} \label{eq:deeplatent_normalized}
\begin{aligned}
    &\textbf{Latent factors:} 
    && z_i \sim \mathrm{N}(0,I), \\[0.5em]
    & \textbf{Texts:} 
    && w_{i,text} \sim NeuralNet(z_i, x_i^c),  \\[0.5em]
    &  \textbf{Images:} 
    && w_{i,images} \sim NeuralNet(z_i, x_i^c),
\end{aligned}
\end{equation*}
\begin{itemize}
    \item[] where:
    
    \begin{itemize}
        \item $z_i$ is a vector of latent variables.
        \item $w_{im}$ is a vector representation for each modality $m$.
        \item The neural networks can take any form (e.g., feedforward, CNNs, or transformers).
    \end{itemize}
\end{itemize}

% A general formulation for a multimodal latent factor model. The factors may be less interpretable (especially in the presence of images).

\end{frame}

\section*{Estimation}

\begin{frame}{}

In general, the marginal likelihood is intractable:

{\footnotesize
\begin{equation*}
p_\theta\big(w_{i,1:M}, y_{i,1:K} \mid x_i\big)
=
\int p_\psi(z_i \mid x_i^p)
\Bigg[ \prod_{m=1}^M p_{\theta_m}\big(w_{im} \mid z_i, x_i^c\big) \Bigg]
\Bigg[ \prod_{k=1}^K p_{\theta_k}\big(y_{ik} \mid z_i, x_i^s\big) \Bigg]
dz_i.
\end{equation*}
}

\vspace{1cm}

\centering \textbf{How do we estimate such a model?}

% There are two ways of thinking about estimation.
% There's a practical recipe view (the autoencoder figures) and a theoretically grounded view (the ELBO that comes after).
    
\end{frame}

\begin{frame}{A Primer on Autoencoders}

\begin{figure}
    \centering
    \includegraphics[width=\linewidth, page=1]{figures/AE.pdf}
\end{figure}

\vspace{-0.5cm}
\begin{equation*}
\begin{aligned}
\mathcal{L}(\theta,\phi; i)=\log p_{\theta}\big(w_{i} \mid q_{\phi}(w_i)\big)
\end{aligned}
\end{equation*}

% Plain autoencoders are deterministic and only aim to minimize the reconstruction loss. They are composed of two neural networks: the encoder and the decoder. You learn the neural network weights by stochastic gradient descent.

% NB: There are potentially many latent factor configurations that can lead to a low reconstruction loss (very unidentified model, which is fine if you don't want to interpret the latent factors, but we typically do). 

\end{frame}

\begin{frame}{A Primer on Variational Autoencoders \citep{kingma2013auto}}

\begin{figure}
    \centering
    \includegraphics[width=\linewidth, page=2]{figures/AE.pdf}
\end{figure}

\vspace{-0.5cm}
\begin{equation*}
\begin{aligned}
\mathcal{L}(\theta,\phi,\psi; i)
&=
\mathbb{E}_{q_\phi}\!\left[\log p_{\theta}\big(w_{i} \mid z_i\big)\right]  -
\mathrm{D}\Big(
q_\phi\big(z_i \mid w_{i}\big)
\;\Vert\;
p_\psi(z_i)
\Big).
\end{aligned}
\end{equation*}

% The VAE adds (i) a stochastic encoder that outputs a distribution over latent variables and (ii) a regularization term that penalizes the model if the distribution of latent factors learned strays too far apart from the prior distribution). This weakly enforces the distribution of latent factors to remain close to the prior, and thus to take a specific form (e.g., Gaussian clusters). It can be theoretically grounded in a maximum likelihood framework (see slides below).

\end{frame}

\begin{frame}{Our Extended Variational Autoencoder}

\begin{figure}
    \centering
    \includegraphics[width=\linewidth]{figures/deeplatent.png}
\end{figure}

\vspace{-0.5cm}
\begin{equation*}
\begin{aligned}
\mathcal{L}(\theta,\phi,\psi; i)
&=
\sum_{m=1}^M 
\mathbb{E}_{q_\phi}\!\left[\log p_{\theta_m}\big(w_{im} \mid z_i, x_i^c\big)\right]
+
\sum_{k=1}^K 
\mathbb{E}_{q_\phi}\!\left[\log p_{\theta_k}\big(y_{ik} \mid z_i, x_i^s\big)\right]
\\[0.2cm]
&\qquad
-
\mathrm{D}\Big(
q_\phi\big(z_i \mid w_{i,1:M}, y_{i,1:K}, x_i\big)
\;\Vert\;
p_\psi(z_i \mid x_i^p)
\Big).
\end{aligned}
\end{equation*}

% We basically extend the VAE to add covariates, outcomes, and multiple modalities. 

% The first sum is: how well do we reconstruct each modality?
% The second sum is: how well do we predict each outcome?
% The third term is the usually penalty to nudge the latent factors to follow a specific distribution.
    
\end{frame}

% The VAE objectives are not heuristic objectives. They are actually grounded in maximum likelihood theory, which we discuss below.

\begin{frame}{Theoretical Foundations: The Evidence Lower Bound (ELBO)}

\small

Define the encoder $q_\phi\big(z_i \mid w_{i,1:M}, y_{i,1:K}, x_i\big)$ from the variational family $\mathcal{Q}_\phi$ (e.g., mean field, full rank gaussian, mixture of gaussians, etc.). 

\vspace{0.3cm} 

Then we can bypass the likelihood and instead maximize the ELBO:
\begin{equation*}
\begin{aligned}
\mathcal{L}(\theta,\phi,\psi; i)
&=
\sum_{m=1}^M 
\mathbb{E}_{q_\phi}\!\left[\log p_{\theta_m}\big(w_{im} \mid z_i, x_i^c\big)\right]
+
\sum_{k=1}^K 
\mathbb{E}_{q_\phi}\!\left[\log p_{\theta_k}\big(y_{ik} \mid z_i, x_i^s\big)\right]
\\[0.2cm]
&\qquad
-
\mathrm{KL}\Big(
q_\phi\big(z_i \mid w_{i,1:M}, y_{i,1:K}, x_i\big)
\;\Vert\;
p_\psi(z_i \mid x_i^p)
\Big).
\end{aligned}
\end{equation*}

The ELBO is a provable lower bound on the marginal likelihood:
\begin{equation*}
\log p_\theta\big(w_{i,1:M}, y_{i,1:K} \mid x_i\big)
\;\ge\;
\mathcal{L}(\theta,\phi,\psi; i).
\end{equation*}

% There are basically a few distinct concepts you need to have clearly in mind: 
% - **the marginal likelihood** is the typical MLE objective: what is the probability of observing this document and these outcomes given the DGP we assumed?
% - **the posterior likelihood** is the probability of z_i conditional on seeing an observation: p(z_i|w,y,x) (it's basically what the encoder q is trying to recover).
% - **the variational family** ==> because the posterior likelihood is intractable, we try to approximate it by specifying a simpler distribution, which we call the variational family. For instance, it is often specified as a Gaussian. You can imagine the true posterior has multiple modes and one unique maximum, and we are simply going to try to approximate this with a Gaussian that has only one mode. In our case, we can parameterize the variational family as a neural network (the encoder). For any given document, the encoder typically outputs a mean and a variance (which define a Gaussian distribution for the latent factor of this observation). We can then further refine this distribution via normalizing flows, which makes the variational family even more flexible, but this is a technical detail.
% - **the ELBO**: given a variational family, the ELBO is defined as a lower bound on the marginal likelihood (you can see that all the expectations are based on the encoder/variational family q). It's a lower bound precisely because we are simplifying the problem by fitting a simpler distribution to approximate the posterior likelihood.

\end{frame}

% OK, so we are maximizing a lower bound on the likelihood, but how 'tight' is this bound (i.e., how closely does it match the likelihood)? The looser it is, the more approximation error we have in our estimation.

\begin{frame}{When is this lower bound tight?}

\small

\textbf{Two sources of approximation error:}
\begin{equation*}
\log p_\theta(d_i \mid x_i)
-
\mathcal{L}(\theta,\phi,\psi; i)
=
\underbrace{V_i}_{\text{variational gap}}
\;+\;
\underbrace{A_i}_{\text{amortization gap}}.
\end{equation*}

\textbf{1. Variational gap $V_i$.}  
This gap exists because the variational family $\mathcal{Q}$ may not contain the true posterior.  
It disappears if:
\begin{equation*}
q_\phi(z_i \mid d_i, x_i) = p_{\theta,\psi}(z_i \mid d_i, x_i) \in \mathcal{Q}.
\end{equation*}

$\rightarrow$ Because normalizing flows are universal approximators of smooth and continuous distributions, adding a flow to a Gaussian encoder allows $V_i$ to be made arbitrarily small \citep{kingma2016improved}.

\vspace{0.5cm}

\textbf{2. Amortization gap $A_i$.}  
This gap exists because the encoder $q_\phi$ must approximate the optimal  
per-data variational solution $q_i^\star$.  
It disappears if:
\begin{equation*}
q_\phi(z_i \mid d_i, x_i) = q_i^\star \quad \text{for all } i.
\end{equation*}

$\rightarrow$ We show that this ideal encoder exists \citep{MargossianBlei2024}. Since neural networks are universal approximators, $A_i$ can be made arbitrarily small.

%In both cases, the remaining difficulty is optimization (i.e., actually finding the optimal solutions).

% There are two sources of approximation error: (i) you specified a variational family that doesn't contain the true posterior and (ii) you are using one big function (the encoder) to approximate the optimal per-data variational solution. 

% This first point refers to the use of a Gaussian to approximate the potentially much more complex posterior likelihood. The solution to this problem is usually to apply a normalizing flow on top of this Gaussian. Normalizing flows are functions that can transform a simple distribution (thee Gaussian in our case) in a much more complicated distribution (hopefully the posterior likelihood!).

% This second point is that, in traditional VI, you do not use an encoder, you just optimize a mean and a variance parameter per data point (so that's A LOT of parameters to optimize over!). We use an encoder, which has two advantages: (i) out of sample predictions are made possible and (ii) we have less parameters to optimize over because the encoder is shared across all observations. The issue is that potentially the encoder outputs suboptimal means and variances relative to traditional VI. In our context, we can prove that the ideal encoder (that maps documents to the optimal per-data traditional VI solution) exists, so a sufficiently expressive encoder should be able, in theory, to find it.

\end{frame}

% From the analysis in the previous slide, we see that we have to be a bit careful in how we define the encoder/variational family. If it's poorly specified, it will lead to a variational gap. This is important when you want to model multiple data sources.

% Basically, when you have multiple modalities, you want each encoder to be able to approximate the posterior likelihood of each modality p(z|w,y,x), and you also want the fusion of these encoders to approximate the posterior likelihood of all these modalities observed jointly p(z|w_1,...,w_M,y,x). The individual encoders allow you to predict out of sample for each modality. The joint fusion allows you to properly estimate the z_i given all the modalities you observe (your estimate of z_i is typically more precise if you observe multiple modalities).

% In our case, we specify the modality-specific encoders and the fusion of these encoders in a way that is theoretically grounded, so that each encoder can really be interpreted as an approximation of the posterior likelihood, and the fusion can capture the posterior likelihood of observing all modalities.

% A nice feature of our approach is that each modality-specific encoder outputs a mean and a precision increment, which literally tells you how much more precise your estimate of z_i is made by observing this modality (so it measures how confident the model is in this modality).

\begin{frame}{How should we encode multiple data sources?}

\small

The true posterior is intractable:
\begin{equation*}
p_{\theta,\psi}(z_i \mid d_i, x_i)
\;\propto\;
p_\psi(z_i \mid x_i^p)\,
\prod_{m \in S_i} p_{\theta_m}(w_{im} \mid z_i, x_i^c)\,
\prod_{k=1}^K p_{\theta_k}(y_{ik} \mid z_i, x_i^s),
\end{equation*}
where $S_i$ is the set of observed modalities and 
$d_i = (w_{i,1:M}, y_{i,1:K})$.

\medskip

The literature proposed a product of experts (PoE) approach \citep{wu2018multimodal}:
\begin{equation*}
q_\phi(z_i \mid d_i, x_i)
\;\propto\;
\prod_{m \in S_i} q_{\phi_m}(z_i \mid w_{im}, y_{i,1:K}, x_i),
\end{equation*}
where $S_i$ is the set of observed modalities and $d_i = (w_{i,1:M}, y_{i,1:K})$. 

\medskip 

Because $q_{\phi_m}(z_i \mid w_{im}, y_{i,1:K}, x_i) \approx p_\psi(z_i \mid x_i^p)  p_{\theta_m, \theta_k}(d_{im} \mid z_i, x_i^c, x_i^s)$,
this inadvertently raises the prior to a power $>1$ and produces an overconfident posterior!

\end{frame}

\begin{frame}{How should we encode multiple data sources?}

\small

The correct form is: 
\begin{equation*}
   q_\phi(z_i \mid d_i, x_i)
        \;\propto\;
        \frac{\prod_{m \in S_i} q_{\phi_m}(z_i \mid w_{im}, y_{i,1:K}, x_i)}
             {\,p_\psi(z_i \mid x_i^p)^{\lvert S_i \rvert - 1}}. 
\end{equation*}

\medskip 

For each modality $m$, our encoder outputs a Gaussian
\[
q_{\phi_m}(z_i \mid w_{im}, y_{i,1:K}, x_i)
= \mathcal{N}\!\big(\mu_{im},\, (\Lambda_{0i} + \Delta_{im})^{-1}\big),
\]
where $\Lambda_{0i}$ is the prior precision (from $p_\psi(z_i \mid x_i^p)$) and $\Delta_{im} \succeq 0$ is a \emph{precision increment} contributed by modality $m$.

\medskip

The fused posterior for a subset of modalities $S_i$ is then:
\begin{align*}
\Lambda_{S_i}
    = \Lambda_{0i} + \sum_{m \in S_i} \Delta_{im} \qquad \text{and} \qquad & \eta_{S_i} 
    = \sum_{m \in S_i} (\Lambda_{0i} + \Delta_{im})\, \mu_{im} \;-\; (|S_i|-1)\Lambda_{0i}\mu_{0i}, \\[0.2cm]
q_\phi(z_i \mid d_i, x_i) 
    &= \mathcal{N}\!\big(\Lambda_{S_i}^{-1}\eta_{S_i},\, \Lambda_{S_i}^{-1}\big).
\end{align*}
    
\end{frame}

\begin{frame}{Estimation Algorithm}

% Subsampling of modalities is key for out-of-sample performance of the encoders. Otherwise, the joint encoder may focus on only one modality and set all other encoders to output minimal or meaningless values. With subsampling, you force the model to treat each modality seriously during training, since it may be the only modality in some minibatches.

% A rule of thumb: you can estimate a topic model on 100,000 documents on a small GPU in under 5 minutes with this approach. For comparison, the STM would take hours if not days to complete its estimation with traditional VI.

\begin{table}[ht!]
    \centering
    \resizebox{0.8\textwidth}{!}{%
    \begin{tabular}{l}
    \toprule
    \textbf{Input:} Dataset $\{(w_{i,1:M}, y_{i,1:K}, x_i)\}_{i=1}^N$ \\
    \textbf{Initialize:} Encoders $\{\phi_m\}_{m=1}^M$; decoders $\{\theta_m\}_{m=1}^M$; predictors $\{\theta_k\}_{k=1}^K$; prior $\psi$. \\
    \midrule
    \textbf{for} epoch $=1,\dots,E$ \textbf{do} \\
    \quad \textbf{for} minibatch $\{(w_{i,1:M}, y_{i,1:K}, x_i)\}_{i=1}^n$ \textbf{do} \\
    \quad\quad Randomly sample modality subset $S_i \subseteq \{1,\ldots,M\}$ for each $i$. \\
    \quad\quad For each $m \in S_i$, compute $q_{\phi_m}(z_i \mid w_{im}, y_{i,1:K}, x_i)$. \\
    \quad\quad Aggregate via PoE: \\
    \quad\quad\quad $\displaystyle q_\phi(z_i \mid d_i, x_i)
        \;\propto\;
        \frac{\prod_{m \in S_i} q_{\phi_m}(z_i \mid w_{im}, y_{i,1:K}, x_i)}
             {\,p_\psi(z_i \mid x_i^p)^{\lvert S_i \rvert - 1}}$. \\
    \quad\quad Sample $\hat{z}_i \sim q_\phi(z_i \mid d_i, x_i)$ via reparameterization. \\
    \quad\quad Evaluate $\log p_{\theta_m}(w_{im}\mid \hat{z}_i, x^c_i)$ for all $m$, and $\log p_{\theta_k}(y_{ik}\mid \hat{z}_i, x^s_i)$ for all $k$. \\
    \quad\quad Compute minibatch ELBO $\widehat{\mathcal{L}}$. \\
    \quad\quad Update $(\{\phi_m\}, \{\theta_m\}, \{\theta_k\}, \psi)$ via SGD on $\widehat{\mathcal{L}}$. \\
    \quad \textbf{end for} \\
    \textbf{end for} \\
    \bottomrule
    \end{tabular}%
    }
\end{table}

\end{frame}

\begin{frame}{Theory vs. Practice}

% This is related to the estimation issues we have been facing. In practice, posterior collapse is very common, and the best solution is to depart from the ELBO and set beta < 1. Still gives good performance in simulations (the simulation you see is actually with beta = 0.01, for instance).

% The trade-off between higher prediction accuracy and lower latent factor interpretability is also interesting in supervised learning contexts.

% You can also mention the WAE here as an alternative to the VAE that has very stable training at the expense of less theoretical guarantees.

\begin{itemize}
    \item Posterior collapse is a common local optimum where the encoder collapses and outputs trivial, uninformative latent factors. It appears when the KL loss imposes too much regularization.
    \item Often, in applications, researchers optimize a weighted ELBO:
    \begin{equation*}
    \begin{aligned}
    \mathcal{L}(\theta,\phi,\psi; i)
    &=
    \sum_{m=1}^M \eta_m
    \mathbb{E}_{q_\phi}\!\left[\log p_{\theta_m}\big(w_{im} \mid z_i, x_i^c\big)\right]
    \\
    &\qquad + 
    \sum_{k=1}^K \eta_k
    \mathbb{E}_{q_\phi}\!\left[\log p_{\theta_k}\big(y_{ik} \mid z_i, x_i^s\big)\right]
    \\[0.2cm]
    &\qquad
    - \beta
    \mathrm{KL}\Big(
    q_\phi\big(z_i \mid w_{i,1:M}, y_{i,1:K}, x_i\big)
    \;\Vert\;
    p_\psi(z_i \mid x_i^p)
    \Big).
    \end{aligned}
    \end{equation*}
    \item Setting $\beta < 1$ typically avoids posterior collapse. Setting $\eta_k > 1$ trades off higher prediction accuracy with lower latent factor interpretability.
    
\end{itemize}

% Transfer learning via pre-trained embeddings (what we do in the third application with LongCLIP)
% Cross-modal encoder-decoder architectures (take texts as input and predict votes, etc.). 
%  Both are more heuristic, less grounded in theory, but probably useful.
    
\end{frame}

\begin{frame}{Application 1: A Structural Topic Model}

The increase in speech polarization in US politics is now a well-established stylized fact {\small (Gentzkow, Shapiro, \& Taddy, 2019)}.

    \begin{figure}
        \centering
        \includegraphics[width = 0.8\textwidth]{figures/gst_leave_out_estimator.png}
    \end{figure}

\centering
\textbf{$\rightarrow$ Is this increase driven by topics or slant?} 

\end{frame}

\begin{frame}{Proposed (Unsupervised) Decomposition Methodology} \label{proposed_methodology}

\begin{itemize}
    
    \item We estimate the GTM on the full \textit{US Congressional Record} (1873 -- 2016). 
    \item We allow party $\times$ session covariates to affect topic choice \textit{and} content.
    \item We use our model of speech to generate samples of partisan speeches.
    \item We simulate 1000 typical speeches for each party $\times$ session under four distinct scenarios:
    \begin{itemize}
        \item \textbf{Total polarization (observed):} Republicans and Democrats differ in topic choice and topical content.
        \item \textbf{Between-topic polarization (topics):} Republicans and Democrats differ in topic choice but not in topical content.
        \item \textbf{Within-topic polarization (slant):} Republicans and Democrats differ in topical content but not in topic choice.
        \item \textbf{Placebo:} Republicans and Democrats do not differ in either dimension.
    \end{itemize}
    \item We compute a speech polarization index per session and scenario.
\end{itemize}

\end{frame}
